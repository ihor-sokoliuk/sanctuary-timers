cmake_minimum_required(VERSION 3.24)
project(SanctuaryTimers VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "This application targets Windows 10/11.")
endif()
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/W4 /utf-8)
else()
  add_compile_options(-Wall -Wextra)
  add_link_options(-static)
endif()
add_compile_definitions(UNICODE _UNICODE NOMINMAX)
add_library(core src/core.cpp)
target_include_directories(core PUBLIC src)
add_library(platform src/platform.cpp)
target_link_libraries(platform PUBLIC core winhttp user32 advapi32)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE core)
add_test(NAME core COMMAND core_tests)
add_executable(platform_tests tests/platform_tests.cpp)
target_link_libraries(platform_tests PRIVATE platform)
add_test(NAME platform COMMAND platform_tests)
add_executable(SanctuaryTimers WIN32 src/main.cpp)
target_link_libraries(SanctuaryTimers PRIVATE platform d2d1 dwrite gdi32 user32 shell32 advapi32 ole32 shlwapi)
if(MINGW)
  target_link_options(SanctuaryTimers PRIVATE -municode)
endif()
